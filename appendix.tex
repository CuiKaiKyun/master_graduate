\section{一阶李导数对状态量偏导}
\label{appendix:lei}
由式\ref{eq:lei_sim}，该矩阵为$12 \times 18$的分块矩阵：
\begin{equation}
    \frac{\partial}{\partial \bm{x}}\big( \mathcal{L}_f^1 \bm{h}(\bm{x}, \bm{u}_k) \big)\Big|_{\bm{x}=\bm{x}_k} = 
    \begin{bmatrix}
        \bm{I}_{3 \times 3} & \bm{A}_{\bm{p} \bm{v}} & \bm{A}_{\bm{p} \bm{\eta}} & \bm{0}_{3 \times 3} & \bm{0}_{3 \times 6} \\
        \bm{0}_{3 \times 3} & \bm{A}_{v v} & \bm{A}_{\bm{v \eta}} & \bm{A}_{\bm{v \omega}} & \bm{A}_{\bm{v \theta}} \\
        \bm{0}_{3 \times 3} & \bm{0}_{3 \times 3} & \bm{A}_{\bm{\eta \eta}} & \bm{A}_{\bm{\eta \omega}} & \bm{0}_{3 \times 6} \\
        \bm{0}_{3 \times 3} & \bm{A}_{\bm{\omega v}} & \bm{A}_{\bm{\omega \eta}} & \bm{A}_{\bm{\omega \omega}} & \bm{A}_{\bm{\omega \theta}}
    \end{bmatrix},
\end{equation}

经整理，这些偏导数分别如下。

位置分量对速度的偏导矩阵$\bm{A}_{\bm{pv}}$满足：
\begin{equation}
    \begin{aligned}
        \bm{A}_{\bm{pv}} =& \frac{\partial \bm{B}_b^e \bm{v}^b \triangle t}{\partial \bm{v}} = \bm{B}_b^e \triangle t \\
        =& \begin{bmatrix}
            \cos\theta\cos\psi & \sin\varphi\sin\theta\cos\psi - \cos\varphi\sin\psi & \cos\varphi\sin\theta\cos\psi + \sin\varphi\sin\psi \\
            \cos\theta\sin\psi & \sin\varphi\sin\theta\sin\psi + \cos\varphi\cos\psi & \cos\varphi\sin\theta\sin\psi - \sin\varphi\cos\psi \\
            -\sin\theta & \sin\varphi\cos\theta & \cos\varphi\cos\theta
        \end{bmatrix} \triangle t.
    \end{aligned}
\end{equation}

位置分量对姿态角的偏导矩阵$\bm{A}_{\bm{p\eta}}$满足：
\begin{equation}
    \bm{A}_{\bm{p\eta}} = \frac{\partial \bm{B}_b^e \bm{v}^b \triangle t}{\partial \bm{\eta}} = \bm{B}_{\bm{p \eta}}\triangle t,
\end{equation}
其中$\bm{B}_{\bm{p \eta}}$满足：
\begin{equation}
    \bm{B}_{\bm{p \eta}} = \frac{\partial \bm{B}_b^e \bm{v}^b}{\partial \bm{\eta}} = 
    \begin{bmatrix}
        B_{\bm{p \eta}}^{11} & B_{\bm{p \eta}}^{12} & B_{\bm{p \eta}}^{13} \\
        B_{\bm{p \eta}}^{21} & B_{\bm{p \eta}}^{22} & B_{\bm{p \eta}}^{23} \\
        B_{\bm{p \eta}}^{31} & B_{\bm{p \eta}}^{32} & B_{\bm{p \eta}}^{33}
    \end{bmatrix},
\end{equation}
且矩阵中的各项元素满足：
\begin{equation}
    \begin{aligned}
        B_{\bm{p \eta}}^{11} =& (\sin\varphi \sin\psi + \cos\varphi \sin\theta \cos\psi)v + (\cos\varphi \sin\psi - \sin\varphi \sin\theta \cos\psi)w, \\
        B_{\bm{p \eta}}^{12} =& (-\sin\theta \cos\psi)u + (\sin\varphi \cos\theta \cos\psi)v + (\cos\varphi \cos\theta \cos\psi)w, \\
        B_{\bm{p \eta}}^{13} =& (-\cos\theta \sin\psi)u + (-\cos\varphi \cos\psi - \sin\varphi \sin\theta \sin\psi)v + (\sin\varphi \cos\psi - \cos\varphi \sin\theta \sin\psi)w, \\
        B_{\bm{p \eta}}^{21} =& (-\sin\varphi \cos\psi + \cos\varphi \sin\theta \sin\psi)v + (-\cos\varphi \cos\psi - \sin\varphi \sin\theta \sin\psi)w\\
        B_{\bm{p \eta}}^{22} =& (-\sin\theta \sin\psi)u + (\sin\varphi \cos\theta \sin\psi)v + (\cos\varphi \cos\theta \sin\psi)w, \\
        B_{\bm{p \eta}}^{23} =& (\cos\theta \cos\psi)u + (-\cos\varphi \sin\psi + \sin\varphi \sin\theta \cos\psi)v + (\sin\varphi \sin\psi + \cos\varphi \sin\theta \cos\psi)w, \\
        B_{\bm{p \eta}}^{31} =& (\cos\varphi \cos\theta)v + (-\sin\varphi \cos\theta)w, \\
        B_{\bm{p \eta}}^{32} =& (-\cos\theta)u + (-\sin\varphi \sin\theta)v + (-\cos\varphi \sin\theta)w, \\
        B_{\bm{p \eta}}^{33} =& 0.
    \end{aligned}
\end{equation}

速度分量对速度的偏导矩阵$\bm{A}_{\bm{vv}}$满足：
\begin{equation}
    \begin{aligned}
        \bm{A}_{\bm{vv}} =& \frac{\partial \bm{v}^e + (m^{-1}\bm{F}^b - \bm{\omega}^b\times\bm{v}^b) \triangle t}{\partial \bm{v}^b} \\
        =& 
        \begin{bmatrix}
            1 - m^{-1} K_N \Omega \triangle t & r \triangle t & -q \triangle t \\
            - \triangle t & 1 - m^{-1} K_N \Omega \triangle t & p \triangle t \\
            q \triangle t & - p \triangle t & 1
        \end{bmatrix}.
    \end{aligned}
\end{equation}

速度分量对姿态角的偏导矩阵$\bm{A}_{\bm{v\eta}}$满足：
\begin{equation}
    \bm{A}_{\bm{v \eta}} = \frac{\partial g \bm{R}_e^b \bm{e}_3 \triangle t}{\partial \bm{v}^b} =
    \begin{bmatrix}
        0 & -\cos\theta & 0 \\
        \cos\varphi \cos\theta & -\sin\varphi sin\theta & 0 \\
        -\sin\varphi \cos\theta & -\cos\varphi \sin\theta & 0
    \end{bmatrix}.
\end{equation}

速度分量对角速度的偏导矩阵$\bm{A}_{\bm{v\omega}}$满足：
\begin{equation}
    \bm{A}_{\bm{v \omega}} = \frac{\partial (-\bm{\omega}^b \times \bm{v}^b) \triangle t}{\partial \bm{\omega}^b} =
    \begin{bmatrix}
        0 & -r & q \\
        r & 0 & -p \\
        -q & p & 0
    \end{bmatrix} \triangle t.
\end{equation}

速度分量对各个参数的偏导矩阵$\bm{A}_{\bm{v\theta}}$满足：
\begin{equation}
    \bm{A}_{\bm{v \theta}} = \frac{\partial m^{-1}\bm{F}^b \triangle t}{\partial \bm{\theta}} = \bm{B}_{\bm{v\theta}} m^{-1} \triangle t,
\end{equation}
其中$\bm{B}_{\bm{v\theta}}$满足：
\begin{equation}
    \bm{B}_{\bm{v\theta}} = \frac{\partial \bm{F}^b }{\partial \bm{\theta}} = [
        \bm{B}_{\bm{v}K_{T0}} \quad
        \bm{B}_{\bm{v}K_{T_J}} \quad
        \bm{B}_{\bm{v}K_{N}} \quad
        \bm{0}_{3 \times 3}
        ],
\end{equation}
且矩阵中的各项元素满足：
\begin{equation}
    \begin{aligned}
        \bm{B}_{\bm{v}K_{T0}} =& [0 \quad 0 \quad -\Omega^2]^T, \\
        \bm{B}_{\bm{v}K_{T_J}} =& [0 \quad 0 \quad -J\Omega^2]^T, \\
        \bm{B}_{\bm{v}K_{N}} =& [-u \Omega \quad -v \Omega \quad 0]^T.
    \end{aligned}
\end{equation}

姿态角分量对姿态角的偏导矩阵$\bm{A}_{\eta\eta}$满足：
\begin{equation}
    \begin{aligned}
        \bm{A}_{\eta\eta} =& \frac{\partial (\bm{\eta} + \bm{Q}\bm{\omega}^b \triangle t)}{\partial \bm{\theta}} = \bm{I}_{3 \times 3} + \frac{\partial \bm{Q}\bm{\omega}^b}{\partial \bm{\theta}}\triangle t \\[6pt]
        =& \bm{I}_{3 \times 3} + 
        \begin{bmatrix}
            \dfrac{\cos\varphi\sin\theta}{\cos\theta} - r\dfrac{\sin\varphi\sin\theta}{\cos\theta}q & \dfrac{\sin\varphi}{\cos^2\theta}q + \dfrac{\cos\varphi}{\cos^2\theta}r & 0 \\[6pt]
            -\sin\varphi q - \cos\varphi r & 0 & 0 \\[6pt]
            \dfrac{\cos\varphi}{\cos\theta}q - \dfrac{\sin\varphi}{\cos\theta}r & \dfrac{\sin\varphi\sin\theta}{\cos^2\theta}q + \dfrac{\cos\varphi\sin\theta}{\cos^2\theta}r & 0\\[6pt]
        \end{bmatrix}.
    \end{aligned}
\end{equation}

姿态角分量对角速度的偏导矩阵$\bm{A}_{\eta\omega}$满足：
\begin{equation}
    \bm{A}_{\eta\omega} = \frac{\partial \bm{Q}\bm{\omega}^b \triangle t}{\partial \bm{\omega}^b} = \bm{Q}\triangle t.
\end{equation}

角速度分量对速度的偏导矩阵$\bm{A}_{\bm{\omega v}}$满足：
\begin{equation}
    \bm{A}_{\bm{\omega v}} = \frac{\partial \bm{J}^{-1} \bm{M}^b \triangle t}{\partial \bm{v}^b} = \frac{\partial \bm{J}^{-1} \bm{M}^b}{\partial \bm{v}^b} \triangle t = \bm{B}_{\bm{\omega v}}\triangle t,
\end{equation}
其中，$\bm{B}_{\bm{\omega v}}$满足：
\begin{equation}
    \bm{B}_{\bm{\omega v}} = \frac{\partial \bm{J}^{-1} \bm{M}^b}{\partial \bm{v}^b} = 
    \begin{bmatrix}
        0 & B_{12} & 0 \\
        B_{21} & 0 & 0 \\
        0 & 0 & 0 \\
    \end{bmatrix},
\end{equation}
且矩阵中的各项元素满足：
\begin{equation}
    \begin{aligned}
        B_{12} =& -\dfrac{(K_{T0} + K_{T_J}J) \Omega K_{l_T} \sin(X_\alpha \alpha_{df}) }{J_x \sin\alpha} - \dfrac{K_{N} \Omega u_r l_N}{J_x}, \\[6pt]
        B_{21} =& \dfrac{(K_{T0} + K_{T_J}J) \Omega K_{l_T} \sin(X_\alpha \alpha_{df}) }{J_y \sin\alpha} + \dfrac{K_{N} \Omega u_r l_N}{J_y}, \\[6pt]
    \end{aligned}
\end{equation}

角速度分量对角速度的偏导矩阵$\bm{A}_{\bm{\omega \omega}}$满足：
\begin{equation}
    \begin{aligned}
        \bm{A}_{\bm{\omega \omega}} =& \frac{\partial \bm{\omega}^b + \bm{J}^{-1} (\bm{M}^b - \bm{\omega}^b \times (\bm{J}\bm{\omega}))\triangle t}{\partial \bm{\omega}^b} \\
        =& \bm{I}_{3 \times 3} + \bm{J}^{-1} \frac{\partial (\bm{M}^b - \bm{\omega}^b \times (\bm{J}\bm{\omega})) }{\partial \bm{\omega}^b} \triangle t = \bm{I}_{3 \times 3} + \bm{J}^{-1} \bm{B}_{\bm{\omega \omega}} \triangle t,
    \end{aligned}
\end{equation}
其中，$\bm{B}_{\bm{\omega \omega}}$满足：
\begin{equation}
    \begin{aligned}
        \bm{B}_{\bm{\omega \omega}} =& \frac{\partial (\bm{M}^b - \bm{\omega}^b \times (\bm{J}\bm{\omega})) }{\partial \bm{\omega}^b} \\
        =& 
        \begin{bmatrix}
            0 & (J_y - J_z) r - J_{fan} \Omega & (J_y - J_z) q \\[8pt]
            (J_z - J_x) r + J_{fan} \Omega & 0 & (J_z - J_x) p \\[8pt]
            (J_x - J_y) q & (J_x - J_y) p & 0
        \end{bmatrix}.
    \end{aligned}
\end{equation}

角速度分量对各参数的偏导矩阵$\bm{A}_{\bm{\omega \theta}}$满足：
\begin{equation}
    \bm{A}_{\bm{\omega \theta}} = \frac{\bm{M}^b}{\partial \bm{\theta}} = \frac{\partial \bm{J}^{-1} \bm{M}^b}{\partial \bm{\theta}} \triangle t = \bm{B}_{\bm{\omega \theta}} \triangle t,
\end{equation}
其中，$\bm{B}_{\bm{\omega \theta}}$满足：
\begin{equation}
    \bm{B}_{\bm{\omega \theta}} = \frac{\partial \bm{J}^{-1} \bm{M}^b}{\partial \bm{\theta}} = [
        \bm{B}_{\bm{v}K_{T0}} \quad
        \bm{B}_{\bm{v}K_{T_J}} \quad
        \bm{B}_{\bm{v}K_{N}} \quad
        \bm{B}_{\bm{v}K_{l_T}} \quad
        \bm{B}_{\bm{v}K_{sta}} \quad
        \bm{B}_{\bm{v}K_{cv}}
    ]
\end{equation}
且矩阵中的各项元素满足：
\begin{equation}
    \begin{aligned}
        \bm{B}_{\bm{v}K_{T0}} =& 
        \biggl[
            -\dfrac{\Omega K_{l_T} \sin(X_\alpha \alpha_{df}) v_r}{J_x \sin\alpha} \quad
            \dfrac{\Omega K_{l_T} \sin(X_\alpha \alpha_{df}) u_r}{J_y \sin\alpha} \quad
            0
        \biggr]^T, \\
        \bm{B}_{\bm{v}K_{T_J}} =& 
        \biggl[
            -\dfrac{J \Omega K_{l_T} \sin(X_\alpha \alpha_{df}) v_r}{J_x \sin\alpha} \quad
            \dfrac{J \Omega K_{l_T} \sin(X_\alpha \alpha_{df}) u_r}{J_y \sin\alpha} \quad
            0
        \biggr]^T, \\
        \bm{B}_{\bm{v}K_{N}} =& 
        \biggl[
            -\Omega v_r l_N \quad
            \Omega u_r l_N \quad
            0
        \biggr]^T, \\
        \bm{B}_{\bm{v}K_{l_T}} =& 
        \biggl[
            -\dfrac{(K_{T0} + K_{T_J}J) \Omega \sin(X_\alpha \alpha_{df}) v_r}{J_x \sin\alpha} \quad
            \dfrac{(K_{T0} + K_{T_J}J) \Omega \sin(X_\alpha \alpha_{df}) u_r}{J_y \sin\alpha} \quad
            0
        \biggr]^T, \\
        \bm{B}_{\bm{v}K_{sta}} =& \biggl[0 \quad 0 \quad -\Omega^2\biggr]^T, \\
        \bm{B}_{\bm{v}K_{cv}} =& \biggl[l_1 (\delta_3 - \delta_1) \quad l_1 (\delta_4 - \delta_2) \quad l_2 (\delta_1 + \delta_2 + \delta_3 + \delta_4)\biggr]^T. \\
    \end{aligned}
\end{equation}
